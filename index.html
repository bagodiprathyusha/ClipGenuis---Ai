function addCalendarPost() { const title = document.getElementById("postTitle").value.trim(); const platform = document.getElementById("postPlatform").value; const date = document.getElementById("postDate").value; if (title === "" || date === "") { alert("Please enter a title and publish date!"); return; } const posts = JSON.parse(localStorage.getItem("calendarPosts") || "[]"); posts.push({ id: Date.now(), title: title, platform: platform, date: date }); localStorage.setItem("calendarPosts", JSON.stringify(posts)); document.getElementById("postTitle").value = ""; document.getElementById("postDate").value = ""; document.getElementById("calendarStatus").innerText = "✓ Post added successfully!"; displayCalendarPosts(); } function displayCalendarPosts() { const posts = JSON.parse(localStorage.getItem("calendarPosts") || "[]"); const list = document.getElementById("calendarList"); list.innerHTML = ""; posts.forEach(function(post) { const card = document.createElement("div"); card.className = "calendar-post"; const title = document.createElement("strong"); title.textContent = post.title; const details = document.createElement("p"); details.textContent = post.platform + " · " + post.date; const deleteButton = document.createElement("button"); deleteButton.textContent = "Delete"; deleteButton.onclick = function() { deleteCalendarPost(post.id); }; card.append(title, details, deleteButton); list.appendChild(card); }); } function deleteCalendarPost(id) { const posts = JSON.parse(localStorage.getItem("calendarPosts") || "[]"); const updatedPosts = posts.filter(function(post) { return post.id !== id; }); localStorage.setItem("calendarPosts", JSON.stringify(updatedPosts)); displayCalendarPosts(); } window.addEventListener("DOMContentLoaded", function() { displayCalendarPosts(); });
